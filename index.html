function doGet(e) { var id = e.parameter.id; var callback = e.parameter.callback; var resposta; if (!id) { resposta = { resultado: 'sem_id' }; } else { var planilha = SpreadsheetApp.getActiveSpreadsheet(); var aba = planilha.getSheetByName('CHECKIN'); var dados = aba.getDataRange().getValues(); resposta = { resultado: 'nao_encontrado' }; for (var i = 1; i < dados.length; i++) { var idPlanilha = dados[i][0]; if (idPlanilha == id) { var nome = dados[i][1]; var status = dados[i][2]; if (status === 'Presente') { var horaAnterior = dados[i][3]; resposta = { resultado: 'ja_confirmado', nome: nome, hora: horaAnterior }; } else { var agora = new Date(); var horaFormatada = Utilities.formatDate(agora, Session.getScriptTimeZone(), 'HH:mm:ss'); aba.getRange(i + 1, 3).setValue('Presente'); aba.getRange(i + 1, 4).setValue(horaFormatada); resposta = { resultado: 'confirmado', nome: nome, hora: horaFormatada }; } break; } } } if (callback) { return ContentService.createTextOutput(callback + '(' + JSON.stringify(resposta) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(resposta)) .setMimeType(ContentService.MimeType.JSON); }
